# The Streamlit dashboard environment, separate from the training environment.
#
# Streamlit brings ~25 packages of its own and constrains future upgrades of
# the core stack, so it stays out of the training environment:
#
#   conda create -n mliam_st python=3.9 pip
#   conda activate mliam_st
#   pip install -r requirements-dashboard.txt
#   pip install -e .
#
# `make dashboard` uses this environment; if yours is named differently,
# run `make dashboard DASHBOARD_ENV=<name>`.
#
# The what-if view runs the trained TFT live, so the dashboard needs the deep
# learning stack too.  It loads and runs the model on the CPU: a CPU-only
# torch wheel is enough here.
